import requests
BASE_URL = "http://127.0.0.1:5000"

def check_endpoint(path):

    url = BASE_URL + path

    print("\nChecking:", url)

    try:

        response = requests.get(
            url,
            timeout=5
        )

        print("Status Code:", response.status_code)

        security_headers = [
            "X-Content-Type-Options",
            "X-Frame-Options",
            "Content-Security-Policy",
            "Strict-Transport-Security"
        ]

        print("\nSecurity Headers:")
        print("----------------------------------------------")

        for header in security_headers:

            if header in response.headers:
                print("[PASS]    ", header)

            else:
                print("[MISSING] ", header)

        if response.status_code == 200:

            try:

                data = response.json()

                if "users" in data:

                    print("\n[WARNING] User data is accessible")
                    print("          without authentication.")

                elif "email" in data:

                    print("\n[WARNING] User information exposed:")
                    print("          Email field is returned.")

            except ValueError:

                print("\nResponse is not JSON.")

    except requests.exceptions.ConnectionError:

        print("\nERROR: API server is not running.")
        print("Start test_api.py first.")

    except requests.exceptions.Timeout:

        print("\nERROR: Connection timed out.")

    except requests.exceptions.RequestException as error:

        print("\nERROR:", error)

def check_login():

    url = BASE_URL + "/api/login"

    print("\nChecking authentication endpoint:")
    print(url)

    data = {
        "username": "wrong-user",
        "password": "wrong-password"
    }

    try:

        response = requests.post(
            url,
            json=data,
            timeout=5
        )

        print("Status Code:", response.status_code)

        if response.status_code == 401:

            print(
                "[PASS] Authentication rejects "
                "invalid credentials."
            )

        else:

            print(
                "[WARNING] Unexpected authentication "
                "response."
            )

    except requests.exceptions.ConnectionError:

        print("\nERROR: API server is not running.")

    except requests.exceptions.RequestException as error:

        print("\nERROR:", error)
